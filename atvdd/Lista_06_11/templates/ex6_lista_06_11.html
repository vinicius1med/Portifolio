class User:
    def __init__(self, name, email, password, age, phone):
        self.__name = name
        self.__email = email
        self.__password = password
        self.__age = age
        self.__phone = phone

    def get_email(self):
        return self.__email

    def get_password(self):
        return self.__password

    def get_name(self):
        return self.__name

registered_users = [
    User("Vinícius", "vinicius@email.com", "1234", 25, "99999-9999"),
    User("Maria", "maria@email.com", "abcd", 30, "88888-8888")
]

session = {}

def authenticate(email, password):
    for user in registered_users:
        if user.get_email() == email and user.get_password() == password:
            session['authenticated_user'] = user.get_name()
            return f"Welcome, {user.get_name()}!"
    return "Invalid email or password."
